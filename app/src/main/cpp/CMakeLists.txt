cmake_minimum_required(VERSION 3.22.1)
project(huanban_import LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(ASSIMP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ASSIMP_TOOLS OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL OFF CACHE BOOL "" FORCE)
set(ASSIMP_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_IMPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_EXPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
foreach(format FBX MMD BLEND OBJ GLTF)
  set(ASSIMP_BUILD_${format}_IMPORTER ON CACHE BOOL "" FORCE)
endforeach()
set(ASSIMP_BUILD_GLTF_EXPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ZLIB ON CACHE BOOL "" FORCE)
set(ASSIMP_IGNORE_GIT_HASH ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_DRACO OFF CACHE BOOL "" FORCE)
set(ASSIMP_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.native-deps/assimp-6.0.2")
if(NOT EXISTS "${ASSIMP_SOURCE}/CMakeLists.txt")
  include(FetchContent)
  FetchContent_Declare(assimp
    URL https://github.com/assimp/assimp/archive/refs/tags/v6.0.2.tar.gz
    URL_HASH SHA256=d1822d9a19c9205d6e8bc533bf897174ddb360ce504680f294170cc1d6319751)
  FetchContent_MakeAvailable(assimp)
else()
  add_subdirectory("${ASSIMP_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/assimp")
endif()
add_library(huanban_import SHARED model_import.cpp)
target_link_libraries(huanban_import PRIVATE assimp log)
target_link_options(huanban_import PRIVATE -Wl,-z,max-page-size=16384)
